CREATE TABLE t_customers (
    c_customer_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    c_customer_name VARCHAR(100) NOT NULL,
    c_customer_city VARCHAR(50) NOT NULL,
    c_customer_join_date DATE DEFAULT CURRENT_DATE,
    c_customer_credit_limit NUMERIC(12,2) CHECK (c_customer_credit_limit >= 0)
);

CREATE TABLE t_orders (
    c_order_id INT GENERATED ALWAYS AS IDENTITY (START WITH 101 INCREMENT BY 1) PRIMARY KEY,
    c_customer_id INT REFERENCES t_customers(c_customer_id) ON DELETE CASCADE,
    c_order_date DATE DEFAULT CURRENT_DATE,
    c_total_amount NUMERIC(12,2) CHECK (c_total_amount >= 0)
);

CREATE TABLE t_products (
    c_product_id INT GENERATED ALWAYS AS IDENTITY (START WITH 201 INCREMENT BY 1) PRIMARY KEY,
    c_product_name VARCHAR(100) NOT NULL,
    c_category VARCHAR(50) NOT NULL,
    c_price NUMERIC(12,2) CHECK (c_price > 0)
);

CREATE TABLE t_order_details (
    c_order_detail_id INT GENERATED ALWAYS AS IDENTITY (START WITH 1 INCREMENT BY 1) PRIMARY KEY,
    c_order_id INT REFERENCES t_orders(c_order_id) ON DELETE CASCADE,
    c_product_id INT REFERENCES t_products(c_product_id),
    c_quantity INT CHECK (c_quantity > 0),
    c_unit_price NUMERIC(12,2) CHECK (c_unit_price > 0)
);
